2. Эндпоинт
python
@app.route('/botodrom/<bot_id>/listen', methods=['POST'])
@require_json
def botodrom_listen(bot_id: str):
    p = get_personality_or_404(bot_id)
    data = request.get_json()
    
    message = data['message']
    tags = data.get('tags', [])
    
    # выбиратор (упрощённо)
    if "кран" in message:
        p.switch_to("entity_1")  # Сантехник
    
    # поиск по профессии
    memories = p.tap_into("сантехника")
    if memories:
        return jsonify({"answer": memories[0]["content"]})
    
    return jsonify({"status": "listening"})
