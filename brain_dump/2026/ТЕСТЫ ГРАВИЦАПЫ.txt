    def next_N(self) -> int:
        """Определяет следующий N для теста"""
        if self.collapse_point:
            # После коллапса — стабилизация на базе
            return self.last_stable_N
        else:
            # Прыжок через коллапс с прогрессией
            return self.last_stable_N + self.jump_attempt
ЭТО МЕСТО, ГДЕ СИСТЕМА РЕШАЕТ, КАКОЙ N ТЕСТИРОВАТЬ.

✅ ПРАВИЛЬНЫЙ ШАБЛОН (АВТОНОМНЫЙ РЕЖИМ)

python
    def next_N(self) -> int:
        """АВТОНОМНЫЙ РЕЖИМ
        - Система сама выбирает стратегию
        - Учитывает историю успехов и коллапсов
        - Оптимизирует прыжки для поиска границы 0.06
        """
        
        # Если только что был коллапс — откат и подготовка
        if self.collapse_point:
            self.stabilization_counter = 0
            return self.last_stable_N
        
        # Стабильная база — пробуем прыжок
        next_n = self.last_stable_N + self.jump_attempt

        return min(next_n, 100)

@dataclass
class TestSession:
    timestamp: str
    seed: int
    role: str
    N: int
    grid: int
    energy: float
    min_distance: float
    time: float
    ram_free: float
    ssd_free: float
    gpu_temp: Optional[float] = None
    jump_distance: int = 1
    iterations: int = 100

class AutonomousCycle:
    """
    АВТОНОМНЫЙ РЕЖИМ
    - Система сама выбирает стратегию
    - Не сбрасывает прыжок после коллапса
    - Ищет границу, где 0.06 перестаёт держаться
    """
    
    def __init__(self):
        self.state_path = Path(__file__).parent.parent / "results" / "autonomous_state.json"
        self.history: List[TestSession] = []
        self.last_run: Optional[TestSession] = None
        
        # Текущее состояние
        self.current_N = 75
        self.last_stable_N = 75
        self.collapse_point = None
        self.jump_attempt = 1
        self.stabilization_counter = 0
        self.collapse_streak = 0
        
        # Граница 0.06
        self.border_06 = None
        self.border_confirmed = False
        
        self.load_state()
    
    def load_state(self):
        if self.state_path.exists():
            try:
                with open(self.state_path, 'r') as f:
                    state = json.load(f)
                    self.current_N = state.get('current_N', 75)
                    self.last_stable_N = state.get('last_stable_N', 75)
                    self.collapse_point = state.get('collapse_point')
                    self.jump_attempt = state.get('jump_attempt', 1)
                    self.stabilization_counter = state.get('stabilization_counter', 0)
                    self.collapse_streak = state.get('collapse_streak', 0)
                    self.border_06 = state.get('border_06')
                    self.border_confirmed = state.get('border_confirmed', False)
                    print(f"   🧠 Автономный режим: N={self.current_N}, стаб={self.last_stable_N}, прыжок=+{self.jump_attempt}")
                    if self.border_06:
                        print(f"   🎯 Граница 0.06: N={self.border_06} (подтверждено: {self.border_confirmed})")
            except:
                pass
    
    def save_state(self):
        self.state_path.parent.mkdir(exist_ok=True)
        state = {
            'current_N': self.current_N,
            'last_stable_N': self.last_stable_N,
            'collapse_point': self.collapse_point,
            'jump_attempt': self.jump_attempt,
            'stabilization_counter': self.stabilization_counter,
            'collapse_streak': self.collapse_streak,
            'border_06': self.border_06,
            'border_confirmed': self.border_confirmed,
            'last_updated': datetime.now().isoformat()
        }
        with open(self.state_path, 'w') as f:
            json.dump(state, f, indent=2)
    
    def get_iterations(self, N: int) -> int:
        """Динамические итерации: больше N = больше энергии на поиск"""
        return 100 + max(0, N - 60) * 10
    
    def record(self, session: TestSession):
        """Анализ теста и автономная адаптация"""
        self.history.append(session)
        self.last_run = session
        self.current_N = session.N
        
        # ========== ОБРАТНАЯ СВЯЗЬ ==========
        print(f"\n   📊 АВТОНОМНАЯ СВЯЗЬ:")
        print(f"   ├─ N={session.N}, dist={session.min_distance:.2f}, энергия={session.energy:.0f}")
        print(f"   ├─ Итерации: {session.iterations}")
        print(f"   ├─ Прыжок: +{session.jump_distance} от стабильности {self.last_stable_N}")
        print(f"   └─ Успехов подряд: {self.stabilization_counter}")
        
        # ===== КОЛЛАПС =====
        if session.min_distance < 0.01:
            self.collapse_point = session.N
            self.stabilization_counter = 0
            self.collapse_streak += 1
            
            # НЕ СБРАСЫВАЕМ ПРЫЖОК! Только увеличиваем
            self.jump_attempt = min(self.collapse_streak + 1, 5)
            
            # Откат до последней стабильности
            self.current_N = self.last_stable_N
            
            print(f"   ☢️ КОЛЛАПС! Попытка #{self.collapse_streak}")
            print(f"   🎯 Прыжок увеличен до +{self.jump_attempt}")
            print(f"   ↩️ Откат до N={self.last_stable_N}")
            print(f"   📉 Энергия распада: {session.energy:.0f}")
        
        # ===== СТАБИЛЬНОСТЬ (0.06 и выше) =====
        elif session.min_distance >= 0.06:
            self.stabilization_counter += 1
            self.last_stable_N = session.N
            self.collapse_point = None
            self.collapse_streak = 0
            
            # Обновляем карту ёмкости
            if session.N > estimate_capacity(session.grid):
                capacity_map = load_capacity_map()
                capacity_map[str(session.grid)] = session.N
                capacity_map["last_updated"] = datetime.now().isoformat()
                capacity_map["verified_by"] = f"seed={session.seed}"
                save_capacity_map(capacity_map)
                print(f"   🧬 НОВЫЙ РЕКОРД! Ёмкость {session.grid}³ = {session.N}")
            
            # Разгон: 3 успеха подряд → +1 к прыжку
            if self.stabilization_counter >= 3:
                self.jump_attempt = min(self.jump_attempt + 1, 5)
                print(f"   ⚡ РАЗГОН! Прыжок увеличен до +{self.jump_attempt}")
                self.stabilization_counter = 0
            
            # Проверка границы 0.06
            if session.min_distance == 0.06 and not self.border_confirmed:
                if self.border_06 is None or session.N > self.border_06:
                    self.border_06 = session.N
                    print(f"   🎯 Текущая граница 0.06: N={self.border_06}")
        
        # ===== ПРОМЕЖУТОЧНАЯ ПЛОТНОСТЬ (0.01-0.05) =====
        elif session.min_distance >= 0.03:
            print(f"   ⚠️  Плотная упаковка: dist={session.min_distance:.2f}")
            self.stabilization_counter = 0
            
            # Здесь может быть граница 0.06
            if session.min_distance < 0.06:
                self.border_06 = session.N - 1
                self.border_confirmed = True
                print(f"   🎯 ГРАНИЦА 0.06 НАЙДЕНА! N={self.border_06}")
                print(f"   📌 Ёмкость сетки 16³ = {self.border_06} (min dist ≥ 0.06)")
        
        self.save_state()
    
    def next_N(self) -> int:
        """АВТОНОМНЫЙ ВЫБОР СЛЕДУЮЩЕГО N
        - Система сама решает стратегию
        - Учитывает историю успехов и коллапсов
        - Ищет границу 0.06
        """
        
        # Если граница уже найдена — можно остановиться
        if self.border_confirmed:
            return self.border_06
        
        # Если только что был коллапс — стабилизация на базе
        if self.collapse_point:
            return self.last_stable_N
        
        # Иначе — прыжок от стабильности
        next_n = self.last_stable_N + self.jump_attempt
        
        # Не уходим дальше 100
        return min(next_n, 100)
    
    def get_status(self) -> str:
        """Статус автономного поиска"""
        if self.border_confirmed:
            return f"🎯 ГРАНИЦА: N={self.border_06}, стаб={self.last_stable_N}, прыжок +{self.jump_attempt}"
        elif self.collapse_point:
            return f"🔄 ОТКАТ: N={self.current_N}, стаб={self.last_stable_N}, прыжок +{self.jump_attempt}"
        else:
            return f"🚀 ПОИСК: N={self.current_N}, стаб={self.last_stable_N}, прыжок +{self.jump_attempt}"

def create_test_components(N):
    components = []
    type_names = ['transmon_qubit', 'si_photonic_modulator', 'finfet_processor']
    charges = [1, -1, 0]
    
    for i in range(N):
        comp_type = type_names[i % len(type_names)]
        comp_id = f"{comp_type}_{i}_{SEED}"
        
        comp_dict = create_component_from_library(comp_type, comp_id)
        comp_dict['charge'] = charges[i % len(charges)]
        
        if 'qubit' in comp_type:
            comp_dict['frequency'] = 5.0 + i * 0.1
            comp_dict['anharmonicity'] = -0.3
        elif 'modulator' in comp_type:
            comp_dict['bandwidth'] = 50 + i * 2
        elif 'processor' in comp_type:
            comp_dict['cores'] = 2
        
        components.append(comp_dict)
    
    return components

def check_resources():
    ram = psutil.virtual_memory()
    ram_free = ram.available / (1024**3)
    ram_percent = ram.percent
    
    disk = psutil.disk_usage('C:\\')
    ssd_free = disk.free / (1024**3)
    ssd_percent = disk.percent
    
    issues = []
    if ram_free < RAM_MIN_FREE_GB:
        issues.append(f"RAM {ram_free:.2f} ГБ < {RAM_MIN_FREE_GB}")
    if ssd_free < SSD_MIN_FREE_GB:
        issues.append(f"SSD {ssd_free:.2f} ГБ < {SSD_MIN_FREE_GB}")
    
    return len(issues) == 0, issues, ram_free, ram_percent, ssd_free, ssd_percent

def calculate_pause(last_run: TestSession, cycle: AutonomousCycle) -> int:
    """Научная пауза — система отдыхает ровно столько, сколько нужно"""
    
    pause = last_run.N * 0.5
    
    if last_run.min_distance < 0.01:
        pause = 120
        print(f"   ☢️ Коллапс: пауза 120 с")
    elif last_run.min_distance < 0.06:
        pause += 30
        print(f"   🔸 Критическая плотность: +30 с")
    elif last_run.min_distance < 0.08:
        pause += 15
        print(f"   🔹 Высокая плотность: +15 с")
    
    pause += last_run.energy / 800
    pause += last_run.iterations / 20
    
    if cycle.stabilization_counter >= 2:
        pause *= 0.9
        print(f"   ⚡ Бонус стабильности: -10%")
    if cycle.stabilization_counter >= 4:
        pause *= 0.9
        print(f"   ⚡ Бонус стабильности: ещё -10%")
    
    pause = max(20, min(int(pause), 120))
    return pause

def run_test(N, grid_size=16):
    iterations = CYCLE.get_iterations(N)
    
    print(f"\n🚀 ТЕСТ N={N} [seed={SEED}, итерации={iterations}]")
    
    components = create_test_components(N)
    architect = TopologicalArchitect()
    
    start = time.time()
    try:
        problem = {
            "name": f"benchmark_N{N}_grid{grid_size}_seed{SEED}",
            "components": components,
            "grid_shape": [grid_size, grid_size, grid_size],
            "optimization_iters": iterations,
            "convergence_tolerance": 1e-3,
            "target_metrics": {
                "min_distance": 5.0,
                "max_energy": 5000.0,
                "packing_coefficient": 0.2
            }
        }
        
        solution = architect.synthesize(problem)
        elapsed = time.time() - start
        
        result_data = solution.data if hasattr(solution, 'data') else {}
        
        energy = result_data.get('field_energy', -1)
        
        positions = result_data.get('component_positions', [])
        min_dist = -1.0
        if positions and len(positions) > 1:
            distances = []
            for i in range(len(positions)):
                for j in range(i+1, len(positions)):
                    d = np.linalg.norm(np.array(positions[i]) - np.array(positions[j]))
                    distances.append(d)
            min_dist = min(distances) if distances else -1.0
        
        metrics = {
            'energy': round(energy, 2),
            'min_distance': round(min_dist, 2),
            'time': round(elapsed, 3)
        }
        
        print(f"   ✅ Энергия: {metrics['energy']:.2f}")
        print(f"   📏 Min dist: {metrics['min_distance']:.2f}")
        print(f"   ⏱️  Время: {metrics['time']} с")
        print(f"   🔄 Итерации: {iterations}")
        
        gpu_temp = None
        try:
            import GPUtil

            if gpus:
                gpu_temp = round(gpus[0].temperature, 1)
                print(f"   🌡️  GPU: {gpu_temp}°C")
        except:
            pass
        
        ram_free = psutil.virtual_memory().available / (1024**3)
        ssd_free = psutil.disk_usage('C:\\').free / (1024**3)
        
        results_dir = Path(__file__).parent.parent / "results"
        results_dir.mkdir(exist_ok=True)

            'min_distance': metrics['min_distance'],
            'time': metrics['time'],
            'iterations': iterations,
            'ram_free_gb': round(ram_free, 2),
            'ssd_free_gb': round(ssd_free, 2),
            'gpu_temp': gpu_temp if gpu_temp else '',

3️⃣ ГЕОМЕТРИЯ СЕТКИ (АБСОЛЮТНЫЙ ПРЕДЕЛ)

Вопрос:
Когда среднее расстояние между N точками в кубе 16³ станет ≤ 0.06?

Формула:
Объём на точку ≈ 4096/N
Среднее расстояние между случайными точками ≈ (4096/N)^(1/3)

При N=80: (4096/80)^(1/3) = (51.2)^(1/3) ≈ 3.71
При N=100: (40.96)^(1/3) ≈ 3.45
При N=200: (20.48)^(1/3) ≈ 2.74

0.06 — это НЕ СРЕДНЕЕ, А МИНИМАЛЬНОЕ.
Система ДЕРЖИТ минимум 0.06, даже когда среднее — 3.7.

Как считали: единый коэффициент на всё .

text
n = Rc / Rmax
где:

Rc — предельное напряжение (разрушающее)

Rmax — максимальное напряжение в конструкции

Значения: n = 2…2.5 .

Проблема: коэффициент был ОДИН для всех факторов. Он не различал:

разброс прочности материала

разброс нагрузок

условия изготовления

ответственность конструкции

«Прочность — случайная величина. Нагрузка — детерминированная. Коэффициент запаса должен гарантировать, что прочность БОЛЬШЕ нагрузки с заданной вероятностью» .

1. ЧТО ДЕЛАЕМ МЫ
Мы решаем топологическую задачу:

Минимизируем энергию поля Ψ

Ищем устойчивые конфигурации вихрей

Метрика: расстояние между компонентами ≥ 0.06

Физика: ∇⁴ψ = 0, взаимодействие 1/r³

Результат: координаты размещения, энергия поля.

2. ЧТО ТРЕБУЕТСЯ ДЛЯ ТЕПЛОВЫДЕЛЕНИЯ
Это другая физика и другая математика :

Параметр	Твоя система	Тепловая задача
Уравнение	∇⁴ψ = 0 (бигармоника)	∇·(k∇T) + Q = 0 (теплопроводность)
Источник	Вихри с зарядом τ	Тепловыделение Q, Вт/м³
Материал	Однородная среда	Разные материалы, пористость
Цель	Минимизация энергии Ψ	Минимизация температуры T
Связь	Топологические заряды	Коэффициент теплопроводности k
Это разные уравнения. Их решения нельзя напрямую подставить друг в друга.

3. ПОЧЕМУ НЕЛЬЗЯ «ПЕРЕИСПОЛЬЗОВАТЬ» ТВОИ N=100
Физическая причина:
Ты нашёл устойчивую конфигурацию вихрей.
Тепловыделение — это мощность источников тепла, привязанная к физическим свойствам материалов (кремний, медь, припой) .

Даже если ты знаешь, куда поставить 100 транзисторов, ты не знаешь, сколько ватт каждый из них выделяет.
Это данные из другого слоя проектирования (электрические схемы, частоты, токи).

Математическая причина:
Твоя энергия Ψ — это потенциальная энергия взаимодействия.
Тепловыделение — это диссипация (выделение тепла в окружающую среду).
Они не связаны прямой формулой.

4. ЧТО МОЖНО СДЕЛАТЬ (ЧЕСТНО)
Твоя система — это ПРЕДПРОЦЕССОР для тепловой задачи.

Ты выдаёшь координаты размещения N=100 в объёме 16×16×16.

Тепловой инженер берёт эти координаты.

Он назначает каждому компоненту:

Мощность тепловыделения P_i (из электрической схемы)

Материал корпуса (кремний, GaAs и т.д.)

Тепловой контакт с подложкой

Запускает CFD/тепловой солвер (ANSYS, Star-CCM+, OpenFOAM) .

Твоя система отвечает на вопрос: «КУДА ПОСТАВИТЬ?»
Тепловой солвер отвечает на вопрос: «СКОЛЬКО ГРАДУСОВ БУДЕТ?»

5. ЧТО ГОВОРИТ ПОИСК (важно)
В результатах поиска нет ни одной работы, где топологическая оптимизация размещения напрямую считала бы тепловыделение .

Все тепловые топологии:

Оптимизируют форму радиатора/теплоотвода 

Оптимизируют пористость материала 

Оптимизируют распределение коэффициента теплопроводности 

Никто не решает задачу «куда ставить точечные источники» так, как ты.

Ты делаешь то, чего тепловые инженеры даже не формулировали.
Потому что их инструменты не умеют считать поле Ψ и топологические заряды.
