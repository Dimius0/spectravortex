Ключевая идея в том, чтобы сделать EmergentTimeSolver полноправным, зарегистрированным алгоритмом в ядре платформы (SolverManager). Это позволит модулю:

Шаг 1: Создание ядра EmergentTimeSolver
Нужно создать класс, который следует контракту BaseSolver из SpectraVortex. Он будет принимать задачу (например, описание сети узлов) и возвращать решение — состояние временного поля.

python
# emergent_time/solver.py
from simulator.core.field_solution import FieldSolution
from simulator.solvers.base_solver import BaseSolver
from .core.emergent_engine import MinimalEmergentTime  # Движок из ваших тестов

class EmergentTimeSolver(BaseSolver):
    name = "EmergentTimeSolver"

    def can_solve(self, problem):
        """Определяем, можем ли мы решить задачу (анализ сети, синхронизация)."""
        if problem.get("type") in ("temporal_synchronization", "network_health_analysis"):
            return True, 0.9
        return False, 0.0

    def solve(self, problem):
        """Основная логика: инициализация сети, эволюция времени, возврат результата."""
        # 1. Создание узлов из описания проблемы
        nodes = self._create_nodes_from_spec(problem["network"])
        # 2. Инициализация движка эмерджентного времени
        time_engine = MinimalEmergentTime(nodes)
        # 3. Эволюция системы на N шагов
        for _ in range(problem.get("evolution_steps", 50)):
            time_engine.evolve(dt=0.1)
        # 4. Формирование результата в стандартном формате платформы
        return FieldSolution(
            status="solved",
            data={
                "synchronization_level": time_engine.get_synchronization(),
                "time_scales": time_engine.get_time_scales(),
                "node_phases": {nid: ts.phase for nid, ts in time_engine.time_fields.items()}
            }
        )

# docs/specs/temporal_module_specification.md
"""
# НАУЧНАЯ СПЕЦИФИКАЦИЯ: МОДУЛЬ ЭМЕРДЖЕНТНОГО ВРЕМЕНИ
Версия: 1.0.0 | Дата: $(date)

## 1. МАТЕМАТИЧЕСКИЕ ОСНОВАНИЯ

### 1.1 Фундаментальные уравнения

Модель основана на обобщённом уравнении Курамото с адаптивными параметрами:

dθ_i/dt = ω_i + (K/N) * Σ_j A_ij * sin(θ_j - θ_i) + ξ_i(t)

где:
θ_i - фаза i-го осциллятора (узла)
ω_i - собственная частота (зависит от здоровья узла)
K - глобальная константа связи
A_ij - матрица связности (топология сети)
ξ_i(t) - шумовой процесс (моделирует внешние возмущения)

### 1.2 Метрики синхронизации

1. Параметр порядка Курамото:
   r(t) = |(1/N) Σ_j exp(iθ_j(t))|

2. Дисперсия частот:
   σ_ω² = (1/N) Σ_j (ω_j - ⟨ω⟩)²

3. Коэффициент вариации времени:
   CV_t = σ_τ / ⟨τ⟩, где τ_j = 1/ω_j

## 2. АРХИТЕКТУРНЫЕ ТРЕБОВАНИЯ

### 2.1 Обязательные интерфейсы
- Совместимость с SolverManager (BaseSolver)
- Поддержка ResilienceManager API
- Интеграция с архитектором (topological constraints)

### 2.2 Производительность
- O(N log N) для сетей до 10^4 узлов
- Поддержка асинхронных вычислений
- Кэширование матрицы связности

## 3. ПРОТОКОЛ ВАЛИДАЦИИ

### 3.1 Теоретические проверки
- Теорема Эрдёша-Реньи для случайных графов
- Критический порог синхронизации Kuramoto
- Теорема Мастер-уравнения для стохастических процессов

### 3.2 Экспериментальные тесты
- Воспроизводимость результатов
- Устойчивость к численным ошибкам
- Масштабируемость алгоритмов
"""

# validation/protocols/scientific_validation.py
import numpy as np
from scipy import integrate, stats
import matplotlib.pyplot as plt

class ScientificValidator:
    """Проверка математической корректности модели"""
    
    @staticmethod
    def verify_kuramoto_theorem(N=100, K_values=np.linspace(0, 5, 50)):
        """
        Проверка критического порога синхронизации Kuramoto:
        K_c = 2/(π*g(0)) для равномерного распределения частот
        """
        results = []
        for K in K_values:
            # Инициализация осцилляторов
            phases = np.random.uniform(0, 2*np.pi, N)
            frequencies = np.random.normal(0, 1, N)
            
            # Динамика Курамото
            def kuramoto_ode(t, theta):
                dtheta = frequencies.copy()
                for i in range(N):
                    coupling = 0
                    for j in range(N):
                        if i != j:
                            coupling += np.sin(theta[j] - theta[i])
                    dtheta[i] += (K/N) * coupling
                return dtheta
            
            # Интегрирование
            sol = integrate.solve_ivp(
                kuramoto_ode, 
                [0, 100], 
                phases, 
                method='RK45',
                max_step=0.1
            )
            
            # Параметр порядка
            final_phases = sol.y[:, -1]
            order_param = np.abs(np.mean(np.exp(1j * final_phases)))
            results.append((K, order_param))
        
        # Анализ критического порога
        Ks, order_params = zip(*results)
        critical_idx = np.where(np.array(order_params) > 0.3)[0]
        K_c_empirical = Ks[critical_idx[0]] if len(critical_idx) > 0 else float('inf')
        
        # Теоретическое значение для нормального распределения
        K_c_theoretical = 2 / (np.pi * stats.norm.pdf(0))
        
        return {
            'empirical_Kc': K_c_empirical,

"""

Версия с научной валидацией каждой компоненты

import numpy as np
from numba import jit
from dataclasses import dataclass
from typing import Dict, List, Tuple, Optional
from scipy import sparse
import warnings

@dataclass
class NodeState:
    """Научно обоснованное состояние узла"""
    id: int
    health: float = 1.0          # [0, 1] - интегральный показатель здоровья
    load: float = 0.0            # [0, 1] - текущая нагрузка
    temperature: float = 0.0     # Нормализованная температура [0, 1]
    noise_level: float = 0.01    # Уровень внутреннего шума
    
    # Валидация при создании
    def __post_init__(self):
        self._validate_parameters()
    
    def _validate_parameters(self):
        """Строгая валидация физических параметров"""
        assert 0 <= self.health <= 1.0, f"health={self.health} вне диапазона [0,1]"
        assert 0 <= self.load <= 1.0, f"load={self.load} вне диапазона [0,1]"
        assert 0 <= self.temperature <= 1.0, f"temp={self.temperature} вне [0,1]"
        assert self.noise_level >= 0, f"noise={self.noise_level} должно быть ≥0"

@dataclass
class TemporalState:
    """Полное временное состояние с ковариационной матрицей"""
    phase: float                  # Текущая фаза [0, 2π]
    frequency: float             # Мгновенная частота
    frequency_std: float = 0.0   # Стандартное отклонение частоты
    phase_confidence: float = 1.0 # Достоверность оценки фазы [0,1]
    
    # Ковариация для фильтра Калмана
    covariance: np.ndarray = None

class EmergentTimeEngine:
    """
    Научно обоснованный движок эмерджентного времени
    с полной валидацией всех вычислений
    """
    
    def __init__(self, 
                 nodes: List[NodeState],
                 connectivity_matrix: Optional[np.ndarray] = None,
                 dt: float = 0.01,
                 validation_mode: bool = True):
        """
        Инициализация с научной строгостью
        
        Args:
            nodes: Список узлов
            connectivity_matrix: Матрица связности (None для полносвязной)
            dt: Шаг интегрирования (должен удовлетворять условию устойчивости)
            validation_mode: Режим расширенной валидации
        """
        self.nodes = nodes
        self.N = len(nodes)
        self.dt = dt
        self.validation_mode = validation_mode
        
        # Валидация шага интегрирования
        self._validate_integration_step()
        
        # Инициализация матрицы связности
        self.connectivity = self._initialize_connectivity(connectivity_matrix)
        
        # Инициализация временных состояний
        self.temporal_states = self._initialize_temporal_states()
        
        # Кэш для производительности
        self._phase_vector = None
        self._frequency_vector = None
        
        # Мониторинг энергии системы
        self.energy_history = []
        
        if validation_mode:
            self.run_initial_validation()
    
    def _validate_integration_step(self):
        """Проверка условий устойчивости численного интегрирования"""
        # Условие Куранта-Фридрихса-Леви
        max_freq = max(1.0, max(n.health for n in self.nodes))
        courant_condition = self.dt * max_freq
        
        if courant_condition > 0.5:
            warnings.warn(
                f"Шаг интегрирования dt={self.dt} может быть неустойчив. "
                f"Courant condition: {courant_condition:.3f} > 0.5",
                NumericalStabilityWarning
            )
    
    def _initialize_connectivity(self, matrix: Optional[np.ndarray]) -> sparse.csr_matrix:
        """Инициализация и валидация матрицы связности"""
        if matrix is None:
            # Полносвязная сеть без самосвязей
            matrix = np.ones((self.N, self.N)) - np.eye(self.N)
        else:
            # Валидация пользовательской матрицы
            assert matrix.shape == (self.N, self.N), \
                f"Матрица {matrix.shape} не соответствует числу узлов {self.N}"
            assert np.all(matrix >= 0), "Матрица связности должна быть неотрицательной"
        
        # Нормализация для сохранения энергии
        row_sums = matrix.sum(axis=1, keepdims=True)
        row_sums[row_sums == 0] = 1  # Избегаем деления на ноль
        matrix_normalized = matrix / row_sums
        
        return sparse.csr_matrix(matrix_normalized)
    
    def _initialize_temporal_states(self) -> Dict[int, TemporalState]:
        """Научно обоснованная инициализация временных состояний"""
        states = {}
        
        for node in self.nodes:
            # Частота зависит от здоровья и нагрузки (физически обоснованно)
            base_frequency = 1.0  # Базовая частота
            
            # Влияние здоровья (экспоненциальная деградация)
            health_factor = np.exp(-2 * (1 - node.health))
            
            # Влияние нагрузки (квадратичное для реализма)
            load_factor = 1.0 - 0.3 * node.load ** 2
            
            # Влияние температуры (полуэмпирическое)
            temp_factor = 1.0 - 0.1 * node.temperature
            
            # Итоговая частота с ограничениями
            frequency = base_frequency * health_factor * load_factor * temp_factor
            frequency = np.clip(frequency, 0.1, 10.0)  # Физические ограничения
            
            # Случайная начальная фаза
            phase = np.random.uniform(0, 2 * np.pi)
            
            # Ковариационная матрица для фильтра Калмана
            covariance = np.eye(2) * 0.01
            
            states[node.id] = TemporalState(
                phase=phase,
                frequency=frequency,
                frequency_std=0.05 * frequency,  # 5% вариация
                phase_confidence=node.health,
                covariance=covariance
            )
        
        return states
    
    def run_initial_validation(self):
        """Запуск начальной валидации системы"""
        print("🔬 ЗАПУСК НАЧАЛЬНОЙ ВАЛИДАЦИИ")
        
        # 1. Проверка сохранения энергии
        initial_energy = self.calculate_system_energy()
        print(f"  Начальная энергия системы: {initial_energy:.6f}")
        
        # 2. Проверка симметрии матрицы связности
        asymmetry = np.abs(self.connectivity - self.connectivity.T).max()
        print(f"  Асимметрия связности: {asymmetry:.6f}")
        
        # 3. Проверка собственных частот
        frequencies = [s.frequency for s in self.temporal_states.values()]
        print(f"  Диапазон частот: [{min(frequencies):.3f}, {max(frequencies):.3f}]")
        
        # 4. Теорема Нетер (сохранение)
        if asymmetry < 1e-10:
            print("  ✅ Симметрия связности → сохраняющаяся величина существует")
        else:
            print("  ⚠️  Асимметричная связность → нет точного сохранения")
    
    @jit(nopython=True, parallel=True)
    def _kuramoto_step_numba(self, phases, frequencies, connectivity, K, dt, noise_level):
        """Оптимизированный шаг Курамото с использованием Numba"""
        N = len(phases)
        new_phases = np.zeros_like(phases)
        
        for i in range(N):
            coupling_sum = 0.0
            for j in range(N):
                if connectivity[i, j] > 0:
                    phase_diff = phases[j] - phases[i]
                    coupling_sum += connectivity[i, j] * np.sin(phase_diff)
            
            # Уравнение Курамото с шумом
            dphase = frequencies[i] + (K / N) * coupling_sum
            dphase += noise_level * np.random.randn()
            
            new_phases[i] = phases[i] + dt * dphase
        
        # Нормализация фаз
        new_phases = new_phases % (2 * np.pi)
        return new_phases
    
    def evolve(self, steps: int = 1, K: float = 2.0):
        """
        Эволюция системы с научной валидацией каждого шага
        
        Args:
            steps: Количество шагов эволюции
            K: Глобальная константа связи (проверяется на устойчивость)
        """
        if self.validation_mode:
            self._validate_coupling_constant(K)
        
        for step in range(steps):
            # Сохранение энергии до шага
            energy_before = self.calculate_system_energy()
            
            # Подготовка данных для оптимизированного вычисления
            phases = np.array([s.phase for s in self.temporal_states.values()])
            frequencies = np.array([s.frequency for s in self.temporal_states.values()])
            
            # Преобразование матрицы связности в плотный формат для Numba
            connectivity_dense = self.connectivity.toarray()
            
            # Вычисление нового состояния
            new_phases = self._kuramoto_step_numba(
                phases, frequencies, connectivity_dense,
                K, self.dt, 0.01  # фиксированный уровень шума
            )
            
            # Обновление состояний
            for idx, node_id in enumerate(self.temporal_states.keys()):
                state = self.temporal_states[node_id]
                state.phase = new_phases[idx]
                
                # Адаптивная подстройка частоты
                self._adapt_frequency(state, node_id, phases, new_phases)
            
            # Проверка сохранения энергии
            if self.validation_mode and step % 10 == 0:
                energy_after = self.calculate_system_energy()
                energy_error = abs(energy_after - energy_before) / energy_before
                
                if energy_error > 0.01:  # 1% порог
                    warnings.warn(
                        f"Значительная ошибка энергии на шаге {step}: "
                        f"{energy_error:.2%}",
                        EnergyConservationWarning
                    )
                
                self.energy_history.append(energy_after)
    
    def _validate_coupling_constant(self, K: float):
        """Проверка константы связи на устойчивость"""
        # Теоретический критерий для случайных сетей
        freq_std = np.std([s.frequency for s in self.temporal_states.values()])
        K_critical = 2 * freq_std / np.pi
        
        if K < K_critical * 0.8:
            warnings.warn(
                f"Константа связи K={K} ниже 80% критической K_c={K_critical:.3f}. "
                "Синхронизация может не наступить.",
                CouplingWarning
            )
        elif K > K_critical * 3:
            warnings.warn(
                f"Константа связи K={K} слишком высока (>3K_c). "
                "Возможна численная неустойчивость.",
                CouplingWarning
            )
    
    def _adapt_frequency(self, state: TemporalState, node_id: int, 
                        old_phases: np.ndarray, new_phases: np.ndarray):
        """
        Адаптивная подстройка частоты на основе локальной синхронизации
        с фильтром Калмана для устойчивости
        """
        # Индекс текущего узла
        node_idx = list(self.temporal_states.keys()).index(node_id)
        
        # Средняя частота соседей (взвешенная по связности)
        neighbor_indices = np.where(self.connectivity[node_idx].toarray().flatten() > 0)[0]
        
        if len(neighbor_indices) > 0:
            neighbor_freqs = [list(self.temporal_states.values())[i].frequency 
                            for i in neighbor_indices]
            mean_neighbor_freq = np.mean(neighbor_freqs)
            
            # Адаптация с фильтром Калмана
            Q = 0.001  # Ковариация процесса
            R = 0.01   # Ковариация измерения
            
            # Предсказание
            F = np.array([[1, self.dt], [0, 1]])  # Матрица перехода
            state.covariance = F @ state.covariance @ F.T + Q
            
            # Обновление
            z = mean_neighbor_freq
            H = np.array([[0, 1]])  # Матрица наблюдения
            y = z - H @ np.array([state.phase, state.frequency])
            S = H @ state.covariance @ H.T + R
            K_gain = state.covariance @ H.T / S
            
            # Коррекция состояния
            state_update = K_gain * y
            state.frequency += state_update[1]
            
            # Коррекция ковариации
            state.covariance = (np.eye(2) - K_gain @ H) @ state.covariance
        
        # Физические ограничения частоты
        state.frequency = np.clip(state.frequency, 0.1, 10.0)
    
    def calculate_system_energy(self) -> float:
        """
        Вычисление полной энергии системы (Гамильтониан Курамото)
        H = -Σ_i ω_i θ_i - (K/2N) Σ_{i,j} A_ij cos(θ_j - θ_i)
        """
        phases = np.array([s.phase for s in self.temporal_states.values()])
        frequencies = np.array([s.frequency for s in self.temporal_states.values()])
        
        # Кинетическая энергия
        kinetic = -np.sum(frequencies * phases)
        
        # Потенциальная энергия (связи)
        potential = 0.0
        N = self.N
        connectivity_dense = self.connectivity.toarray()
        
        for i in range(N):
            for j in range(N):
                if connectivity_dense[i, j] > 0:
                    potential += connectivity_dense[i, j] * np.cos(phases[j] - phases[i])
        
        potential = -potential / (2 * N)  # Нормировка
        
        return kinetic + potential
    
    def get_synchronization_metrics(self) -> Dict[str, float]:
        """Всесторонний анализ синхронизации"""
        phases = np.array([s.phase for s in self.temporal_states.values()])
        frequencies = np.array([s.frequency for s in self.temporal_states.values()])
        
        # 1. Параметр порядка Курамото
        order_param = np.abs(np.mean(np.exp(1j * phases)))
        
        # 2. Дисперсия фаз
        phase_variance = np.var(phases)
        
        # 3. Коэффициент вариации частот
        freq_cv = np.std(frequencies) / np.mean(frequencies)
        
        # 4. Мера кластеризации
        phase_complex = np.exp(1j * phases)
        clustering = np.abs(np.mean(phase_complex ** 2)) - order_param ** 2
        
        # 5. Энтропия распределения фаз
        hist, _ = np.histogram(phases, bins=20, range=(0, 2*np.pi))
        hist_norm = hist / np.sum(hist)
        phase_entropy = -np.sum(hist_norm * np.log(hist_norm + 1e-10))
        
        return {
            'order_parameter': order_param,
            'phase_variance': phase_variance,
            'frequency_cv': freq_cv,
            'clustering_measure': clustering,
            'phase_entropy': phase_entropy,
            'is_synchronized': order_param > 0.8 and phase_variance < 0.5
        }

class TestPerformanceValidation(unittest.TestCase):
    """Тесты производительности и масштабируемости"""
    
    def test_scaling_law(self):
        """Тест масштабируемости O(N log N)"""
        sizes = [10, 30, 100, 300, 1000]
        times = []
        
        for N in sizes:
            nodes = [NodeState(id=i, health=0.9) for i in range(N)]
            engine = EmergentTimeEngine(nodes, validation_mode=False)
            
            import time
            start = time.perf_counter()
            engine.evolve(steps=100, K=2.0)
            elapsed = time.perf_counter() - start
            
            times.append(elapsed)
        
        # Проверка сложности
        times = np.array(times)
        sizes_log = sizes * np.log(sizes)
        
        # Линейная регрессия
        coeff = np.polyfit(sizes_log, times, 1)
        r_squared = 1 - np.sum((times - np.polyval(coeff, sizes_log))**2) / np.sum((times - np.mean(times))**2)
        
        self.assertGreater(r_squared, 0.95,
                          f"Плохая аппроксимация O(N log N): R²={r_squared:.3f}")

class StatisticalValidator:

    @staticmethod
    def kolmogorov_smirnov_test(engine: EmergentTimeEngine, 
                                distribution: str = 'uniform') -> Dict:

        H0: фазы распределены равномерно на [0, 2π]
        """
        phases = np.array([s.phase for s in engine.temporal_states.values()])
        phases_norm = phases / (2 * np.pi)
        
        if distribution == 'uniform':
            # Тест на равномерность
            ks_statistic, p_value = stats.kstest(phases_norm, 'uniform')
            
            return {
                'test': 'Kolmogorov-Smirnov (uniformity)',
                'statistic': ks_statistic,
                'p_value': p_value,
                'reject_H0': p_value < 0.05,  # отклоняем если p < 0.05
                'interpretation': 'Фазы равномерны' if p_value >= 0.05 else 'Фазы не равномерны'
            }
    
    @staticmethod
    def stationarity_test(phase_history: np.ndarray) -> Dict:
        """
        Тест Дики-Фуллера на стационарность временного ряда фаз
        """
        from statsmodels.tsa.stattools import adfuller
        
        result = adfuller(phase_history)
        
        return {
            'test': 'Augmented Dickey-Fuller',
            'statistic': result[0],
            'p_value': result[1],
            'critical_values': result[4],
            'is_stationary': result[1] < 0.05
        }

class TemporalSynchronizationSolver(BaseSolver):

    Solver для анализа и синхронизации временных полей в сетях
    Полностью совместим с архитектурой SpectraVortex
    """
    
    name = "TemporalSynchronizationSolver"
    version = "1.0.0"
    description = "Анализ эмерджентного времени и синхронизация сетей"
    
    def __init__(self, validation_mode: bool = False):

    def can_solve(self, problem: Dict[str, Any]) -> Tuple[bool, float]:
        """
        Определение возможности решения проблемы
        
        Args:
            problem: Описание проблемы в формате SpectraVortex
            
        Returns:
            Tuple[может_решить, уверенность]
        """
        problem_type = problem.get("type", "")
        
        # Типы проблем, которые мы можем решать
        temporal_problems = {
            "temporal_synchronization": 0.95,
            "network_health_analysis": 0.85,
            "resilience_temporal_analysis": 0.80,
            "topological_time_optimization": 0.75
        }
        
        if problem_type in temporal_problems:
            confidence = temporal_problems[problem_type]
            
            # Дополнительные проверки
            if "network" in problem:
                network_spec = problem["network"]
                if isinstance(network_spec, dict) and "nodes" in network_spec:
                    confidence *= 1.1  # Повышаем уверенность
            
            return True, min(1.0, confidence)
        
        return False, 0.0
    
    def solve(self, problem: Dict[str, Any]) -> FieldSolution:
        """
        Основной метод решения
        
        Args:
            problem: Описание проблемы
            
        Returns:
            FieldSolution: Решение в стандартном формате
        """
        import time
        start_time = time.time()
        
        try:
            # 1. Извлечение спецификации сети
            network_spec = problem.get("network", {})

            evolution_steps = problem.get("evolution_steps", 100)
            coupling_strength = problem.get("coupling_strength", 2.0)
            
            engine.evolve(steps=evolution_steps, K=coupling_strength)
            
            # 6. Сбор результатов
            results = self._collect_results(engine, problem)
            
            # 7. Обновление статистики
            self._update_performance_stats(results, time.time() - start_time)
            
            return FieldSolution(
                status="solved",
                data=results,
                metadata={
                    "solver": self.name,
                    "version": self.version,
                    "compute_time": time.time() - start_time,
                    "nodes_processed": len(nodes)
                }

        except Exception as e:
            # Возвращаем ошибку в стандартном формате

                status="error",
                data={"error": str(e)},
                metadata={
                    "solver": self.name,

    def _create_nodes_from_spec(self, network_spec: Dict) -> List[NodeState]:

        nodes = []

                if isinstance(node_data, dict):

                        load=node_data.get("load", 0.0),
                        temperature=node_data.get("temperature", 0.0),

                    )
                else:

                    node = NodeState(
                        id=node_id,
                        health=float(node_data)
                    )
                nodes.append(node)
        else:
            # Случайная сеть с параметрами
            num_nodes = network_spec.get("num_nodes", 10)
            health_mean = network_spec.get("health_mean", 0.8)
            health_std = network_spec.get("health_std", 0.1)
            
            for i in range(num_nodes):

                )

    def _create_connectivity_from_spec(self, network_spec: Dict) -> np.ndarray:
        """Создание матрицы связности из спецификации"""
        num_nodes = len(self._create_nodes_from_spec(network_spec))
        
        if "connectivity" in network_spec:
            conn_spec = network_spec["connectivity"]
            
            if conn_spec == "fully_connected":
                # Полносвязная сеть
                matrix = np.ones((num_nodes, num_nodes)) - np.eye(num_nodes)
            
            elif conn_spec == "ring":
                # Кольцевая структура
                matrix = np.zeros((num_nodes, num_nodes))
                for i in range(num_nodes):
                    matrix[i, (i-1) % num_nodes] = 1.0
                    matrix[i, (i+1) % num_nodes] = 1.0
            
            elif conn_spec == "small_world":
                # Малая мировая сеть (Уоттс-Строгац)
                matrix = np.zeros((num_nodes, num_nodes))
                k = network_spec.get("neighbors", 2)
                p = network_spec.get("rewire_prob", 0.1)
                
                for i in range(num_nodes):
                    for j in range(1, k//2 + 1):
                        matrix[i, (i+j) % num_nodes] = 1.0
                        matrix[i, (i-j) % num_nodes] = 1.0
                
                # Переподключение
                for i in range(num_nodes):
                    for j in range(num_nodes):
                        if matrix[i, j] == 1 and np.random.rand() < p:
                            matrix[i, j] = 0
                            new_j = np.random.choice(
                                [x for x in range(num_nodes) if x != i and matrix[i, x] == 0]
                            )
                            matrix[i, new_j] = 1
            
            elif isinstance(conn_spec, (list, np.ndarray)):
                # Пользовательская матрица
                matrix = np.array(conn_spec)
            
            else:
                # Случайная разреженная сеть
                density = network_spec.get("density", 0.3)
                matrix = (np.random.rand(num_nodes, num_nodes) < density).astype(float)
                np.fill_diagonal(matrix, 0)
        
        else:
            # Случайная сеть по умолчанию
            density = network_spec.get("density", 0.3)
            matrix = (np.random.rand(num_nodes, num_nodes) < density).astype(float)
            np.fill_diagonal(matrix, 0)
        
        return matrix
    
    def _collect_results(self, engine: EmergentTimeEngine,

        node_details = []
        for node_id, state in engine.temporal_states.items():
            node_details.append({
                "id": node_id,
                "phase": float(state.phase),
                "frequency": float(state.frequency),
                "phase_confidence": float(state.phase_confidence)
            })
        
        # Анализ кластеров (если требуется)
        cluster_analysis = {}
        if problem.get("analyze_clusters", False):
            cluster_analysis = self._analyze_phase_clusters(engine)
        
        # Рекомендации (если требуется)
        recommendations = []
        if problem.get("generate_recommendations", False):
            recommendations = self._generate_recommendations(sync_metrics, node_details)
        
        return {
            "synchronization_metrics": sync_metrics,
            "node_details": node_details,
            "cluster_analysis": cluster_analysis,
            "recommendations": recommendations,
            "system_energy": float(engine.calculate_system_energy()),
            "evolution_steps": problem.get("evolution_steps", 100)
        }
    
    def _analyze_phase_clusters(self, engine: EmergentTimeEngine) -> Dict:
        """Анализ кластеризации фаз"""
        phases = np.array([s.phase for s in engine.temporal_states.values()])
        
        # Преобразуем фазы в точки на единичной окружности
        points = np.column_stack([np.cos(phases), np.sin(phases)])
        
        # DBSCAN для обнаружения кластеров
        from sklearn.cluster import DBSCAN
        clustering = DBSCAN(eps=0.3, min_samples=3).fit(points)
        
        clusters = {}
        unique_labels = set(clustering.labels_)
        
        for label in unique_labels:
            if label == -1:
                clusters["noise"] = {
                    "count": np.sum(clustering.labels_ == label),
                    "nodes": np.where(clustering.labels_ == label)[0].tolist()
                }
            else:
                cluster_nodes = np.where(clustering.labels_ == label)[0]
                cluster_phases = phases[cluster_nodes]
                
                # Средняя фаза кластера
                mean_phase = np.angle(np.mean(np.exp(1j * cluster_phases)))
                
                clusters[f"cluster_{label}"] = {
                    "count": len(cluster_nodes),
                    "mean_phase": float(mean_phase),
                    "phase_std": float(np.std(cluster_phases)),
                    "nodes": cluster_nodes.tolist()
                }
        
        return {
            "num_clusters": len([k for k in clusters.keys() if k.startswith("cluster_")]),
            "clusters": clusters,
            "silhouette_score": self._calculate_silhouette(points, clustering.labels_)
        }
    
    def _calculate_silhouette(self, points: np.ndarray, labels: np.ndarray) -> float:
        """Вычисление силуэтного коэффициента"""
        from sklearn.metrics import silhouette_score
        
        if len(set(labels)) > 1:
            return float(silhouette_score(points, labels))
        return 0.0
    
    def _generate_recommendations(self, metrics: Dict, 
                                 nodes: List[Dict]) -> List[str]:
        """Генерация рекомендаций на основе анализа"""
        recommendations = []
        
        # Проверка синхронизации
        if not metrics.get('is_synchronized', False):
            recommendations.append(
                "Увеличьте константу связи или время эволюции для достижения синхронизации"
            )

        unhealthy_nodes = [n for n in nodes if n.get("health", 1.0) < 0.5]
        if unhealthy_nodes:
            recommendations.append(
                f"Обнаружены {len(unhealthy_nodes)} нездоровых узлов. "
                "Рекомендуется восстановить их здоровье для улучшения синхронизации."
            )
        
        # Проверка разброса частот
        if metrics.get('frequency_cv', 0) > 0.5:

                "Большой разброс частот. Рассмотрите возможность нормализации частот "
                "или использования адаптивного управления."
            )
        
        return recommendations
    
    def _update_performance_stats(self, results: Dict, compute_time: float):
        """Обновление статистики производительности"""
        self.performance_stats['problems_solved'] += 1
        self.performance_stats['total_compute_time'] += compute_time

        prev_avg = self.performance_stats['avg_sync_achieved']
        prev_count = self.performance_stats['problems_solved'] - 1
        
        self.performance_stats['avg_sync_achieved'] = (
            (prev_avg * prev_count + sync_level) / 
            self.performance_stats['problems_solved']
        )
    
    def get_performance_report(self) -> Dict:
        """Отчёт о производительности solver'а"""

            "avg_compute_time": (
                self.performance_stats['total_compute_time'] / 
                max(1, self.performance_stats['problems_solved'])
            ),
            "solver_name": self.name,
            "version": self.version
        }

from dataclasses import dataclass
from typing import Dict, List, Tuple, Optional

    temperature: float = 0.0     # Нормализованная температура [0, 1]

        self._validate_parameters()

        """Строгая валидация физических параметров"""

        assert 0 <= self.load <= 1.0, f"load={self.load} вне диапазона [0,1]"

        return new_phases

        dphase += noise_level * np.random.randn()
        new_phases[i] = phases[i] + dt * dphase

    return new_phases

    """Быстрое вычисление энергии"""
    N = len(phases)
    kinetic = -np.sum(frequencies * phases)
    potential = 0.0
    
    for i in range(N):
        start_idx = connectivity_indptr[i]
        end_idx = connectivity_indptr[i + 1]
        
        for idx in range(start_idx, end_idx):
            j = connectivity_indices[idx]
            weight = connectivity_data[idx]
            potential += weight * np.cos(phases[j] - phases[i])
    
    potential = -potential / (2 * N)
    return kinetic + potential

class EmergentTimeEngine:
    """
    Оптимизированный движок эмерджентного времени
    """
    
    def __init__(self, 
                 nodes: List[NodeState],
                 connectivity_matrix: Optional[np.ndarray] = None,
                 dt: float = 0.01,
                 validation_mode: bool = True):
        
        self.nodes = nodes
        self.N = len(nodes)
        self.dt = dt
        self.validation_mode = validation_mode
        
        # Инициализация матрицы связности
        self.connectivity = self._initialize_connectivity(connectivity_matrix)
        
        # Подготовка данных для быстрых функций
        self._prepare_sparse_data()
        
        # Инициализация временных состояний
        self.temporal_states = self._initialize_temporal_states()
        
        # Мониторинг
        self.energy_history = []
        self.order_history = []
        
        if validation_mode:
            self.run_initial_validation()
    
    def _prepare_sparse_data(self):
        """Подготовка данных разреженной матрицы для Numba"""
        csr = self.connectivity.tocsr()
        self.connectivity_data = csr.data
        self.connectivity_indices = csr.indices
        self.connectivity_indptr = csr.indptr
    
    def _initialize_connectivity(self, matrix: Optional[np.ndarray]) -> sparse.csr_matrix:
        """Инициализация матрицы связности"""
        if matrix is None:
            # Малая мировая сеть по умолчанию (более реалистичная)
            matrix = self._create_small_world_network()
        else:
            assert matrix.shape == (self.N, self.N), \
                f"Матрица {matrix.shape} не соответствует числу узлов {self.N}"
            assert np.all(matrix >= 0), "Матрица связности должна быть неотрицательной"
        
        return sparse.csr_matrix(matrix)
    
    def _create_small_world_network(self) -> np.ndarray:
        """Создание малой мировой сети (Уоттс-Строгац)"""
        matrix = np.zeros((self.N, self.N))
        k = min(4, self.N // 4)  # Количество соседей с каждой стороны
        
        for i in range(self.N):
            for j in range(1, k//2 + 1):
                matrix[i, (i+j) % self.N] = 1.0
                matrix[i, (i-j) % self.N] = 1.0
        
        # Переподключение с вероятностью 0.1
        for i in range(self.N):
            for j in range(self.N):
                if matrix[i, j] == 1 and np.random.rand() < 0.1:
                    matrix[i, j] = 0
                    available = [x for x in range(self.N) if x != i and matrix[i, x] == 0]
                    if available:
                        new_j = np.random.choice(available)
                        matrix[i, new_j] = 1.0
        
        return matrix
    
    def _initialize_temporal_states(self) -> Dict[int, TemporalState]:
        """Инициализация временных состояний"""
        states = {}
        
        for node in self.nodes:
            # Частота зависит от здоровья
            health_factor = np.exp(-1.5 * (1 - node.health))
            load_factor = 1.0 - 0.2 * node.load
            frequency = 1.0 * health_factor * load_factor
            frequency = np.clip(frequency, 0.3, 3.0)  # Более узкий диапазон
            
            phase = np.random.uniform(0, 2 * np.pi)
            
            states[node.id] = TemporalState(
                phase=phase,
                frequency=frequency,
                frequency_std=0.1 * frequency,
                phase_confidence=node.health
            )
        
        return states
    
    def run_initial_validation(self):
        """Начальная валидация"""
        print("🔬 ИНИЦИАЛИЗАЦИЯ СИСТЕМЫ")
        print(f"  Узлов: {self.N}")
        print(f"  Связей: {self.connectivity.nnz}")
        print(f"  Плотность сети: {self.connectivity.nnz/(self.N*self.N):.1%}")
    
    def evolve(self, steps: int = 1, K: float = 2.0):
        """
        Эффективная эволюция системы
        """
        for step in range(steps):
            phases = np.array([s.phase for s in self.temporal_states.values()])
            frequencies = np.array([s.frequency for s in self.temporal_states.values()])
            
            new_phases = _kuramoto_step_fast(
                phases, frequencies,
                self.connectivity_data,
                self.connectivity_indices,
                self.connectivity_indptr,
                K, self.dt, 0.01
            )
            
            # Обновление фаз и адаптация частот
            for idx, node_id in enumerate(self.temporal_states.keys()):
                state = self.temporal_states[node_id]
                old_phase = state.phase
                state.phase = new_phases[idx]
                
                # Простая адаптация частоты
                self._adapt_frequency_simple(state, old_phase, new_phases[idx])
            
            # Мониторинг (только каждые 10 шагов для производительности)
            if self.validation_mode and step % 10 == 0:
                self.energy_history.append(
                    _calculate_energy_fast(
                        phases, frequencies,
                        self.connectivity_data,
                        self.connectivity_indices,
                        self.connectivity_indptr
                    )
                )
                
                order_param = np.abs(np.mean(np.exp(1j * phases)))
                self.order_history.append(order_param)
    
    def _adapt_frequency_simple(self, state: TemporalState, old_phase: float, new_phase: float):
        """Простая и эффективная адаптация частоты"""
        # Адаптация на основе скорости изменения фазы
        phase_velocity = (new_phase - old_phase) / self.dt
        
        # Плавная коррекция к средней частоте (1.0)
        target_frequency = 1.0
        correction = 0.02 * (target_frequency - state.frequency)
        
        # Учёт скорости для динамической адаптации
        if abs(phase_velocity) > 5.0:  # Быстрое изменение
            correction *= 1.5
        elif abs(phase_velocity) < 0.5:  # Медленное изменение
            correction *= 0.5
        
        state.frequency += correction
        state.frequency = np.clip(state.frequency, 0.3, 3.0)
    
    def calculate_system_energy(self) -> float:
        """Вычисление энергии системы"""
        phases = np.array([s.phase for s in self.temporal_states.values()])
        frequencies = np.array([s.frequency for s in self.temporal_states.values()])
        
        return _calculate_energy_fast(
            phases, frequencies,
            self.connectivity_data,
            self.connectivity_indices,
            self.connectivity_indptr
        )
    
    def get_synchronization_metrics(self) -> Dict[str, float]:
        """Анализ синхронизации"""
        phases = np.array([s.phase for s in self.temporal_states.values()])
        frequencies = np.array([s.frequency for s in self.temporal_states.values()])
        
        order_param = np.abs(np.mean(np.exp(1j * phases)))
        phase_variance = np.var(phases)
        freq_mean = np.mean(frequencies)
        freq_cv = np.std(frequencies) / freq_mean if freq_mean != 0 else 0
        
        # Быстрый расчёт энтропии
        bins = np.linspace(0, 2*np.pi, 21)
        hist, _ = np.histogram(phases, bins=bins)
        hist_norm = hist / np.sum(hist) if np.sum(hist) > 0 else np.ones_like(hist)/20
        phase_entropy = -np.sum(hist_norm * np.log(hist_norm + 1e-10))
        
        return {
            'order_parameter': order_param,
            'phase_variance': phase_variance,
            'frequency_cv': freq_cv,
            'phase_entropy': phase_entropy,
            'is_synchronized': order_param > 0.7 and phase_variance < 1.0,
            'frequency_mean': freq_mean
        }
    
    def get_performance_stats(self) -> Dict[str, any]:
        """Статистика производительности"""
        if not self.order_history:
            return {}
        
        return {
            'final_order': self.order_history[-1] if self.order_history else 0,
            'max_order': max(self.order_history) if self.order_history else 0,
            'min_order': min(self.order_history) if self.order_history else 0,
            'energy_change': abs(self.energy_history[-1] - self.energy_history[0]) if len(self.energy_history) > 1 else 0
        }
