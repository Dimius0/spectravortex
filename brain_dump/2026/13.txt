#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
safe_benchmark.py — МИНИМАЛЬНАЯ ВЕРСИЯ
Только защита памяти/SSD + запуск одного теста
"""

import os
import sys
import time
import csv
from pathlib import Path
from datetime import datetime

# ========== ПОРОГИ ==========
RAM_MIN_FREE_GB = 1.2
SSD_MIN_FREE_GB = 5.0
PAUSE_BETWEEN_TESTS_SEC = 60
# ============================

try:
    import psutil
except ImportError:
    print("❌ Установи psutil: pip install psutil")
    sys.exit(1)

sys.path.insert(0, str(Path(__file__).parent.parent))
from spectravortex import TopologicalArchitect
from spectravortex.components import Qubit, Modulator, ClassicalProcessor

def check_resources():
    ram_free = psutil.virtual_memory().available / (1024**3)
    ssd_free = psutil.disk_usage('C:\\').free / (1024**3)
    
    issues = []
    if ram_free < RAM_MIN_FREE_GB:
        issues.append(f"RAM {ram_free:.2f} ГБ < {RAM_MIN_FREE_GB}")
    if ssd_free < SSD_MIN_FREE_GB:
        issues.append(f"SSD {ssd_free:.2f} ГБ < {SSD_MIN_FREE_GB}")
    
    return len(issues) == 0, issues, ram_free, ssd_free

def run_test(N=10):
    print(f"\n🚀 Тест N={N}")
    
    components = []
    for i in range(N):
        if i % 3 == 0:
            components.append(Qubit(charge=+1, frequency=5.0))
        elif i % 3 == 1:
            components.append(Modulator(charge=-1, bandwidth=50))
        else:
            components.append(ClassicalProcessor(charge=0, cores=2))
    
    architect = TopologicalArchitect(
        grid_shape=(16, 16, 16),
        max_iterations=50,
        convergence_tolerance=1e-3
    )
    
    start = time.time()
    solution = architect.optimize(components)
    elapsed = time.time() - start
    
    print(f"   ✅ Энергия: {solution.energy:.2f}")
    print(f"   ⏱️  Время: {elapsed:.2f} с")
    return solution

def main():
    print("🔬 SpectraVortex — защищённый тест")
    
    ok, issues, ram, ssd = check_resources()
    if not ok:
        print(f"❌ Ресурсы: {', '.join(issues)}")
        return
    
    print(f"✅ RAM: {ram:.2f} ГБ, SSD: {ssd:.2f} ГБ")
    run_test(N=10)
    
if __name__ == "__main__":
    main()

from architect import TopologicalArchitect
from architect.components import Qubit, Modulator, ClassicalProcessor
# ============================================

def check_resources():
    """Проверка памяти и диска перед запуском"""
    ram = psutil.virtual_memory()
    ram_free = ram.available / (1024**3)
    ram_percent = ram.percent
    
    disk = psutil.disk_usage('C:\\')
    ssd_free = disk.free / (1024**3)
    ssd_percent = disk.percent
    
    issues = []
    if ram_free < RAM_MIN_FREE_GB:
        issues.append(f"RAM {ram_free:.2f} ГБ < {RAM_MIN_FREE_GB}")
    if ssd_free < SSD_MIN_FREE_GB:
        issues.append(f"SSD {ssd_free:.2f} ГБ < {SSD_MIN_FREE_GB}")
    
    return len(issues) == 0, issues, ram_free, ram_percent, ssd_free, ssd_percent

def run_test(N, grid_size=16):
    """Один прогон архитектора"""
    print(f"\n🚀 Тест N={N} (сетка {grid_size}³)")
    
    # Создаём компоненты
    components = []
    for i in range(N):
        if i % 3 == 0:
            components.append(Qubit(charge=+1, frequency=5.0 + i*0.1))
        elif i % 3 == 1:
            components.append(Modulator(charge=-1, bandwidth=50 + i*2))
        else:
            components.append(ClassicalProcessor(charge=0, cores=2))
    
    # Создаём архитектор
    architect = TopologicalArchitect(
        grid_shape=(grid_size, grid_size, grid_size),
        max_iterations=100,          # Уменьшено для скорости
        convergence_tolerance=1e-3   # Пониженная точность (бережём CPU)
    )
    
    start = time.time()
    try:
        solution = architect.optimize(
            components=components,
            objective='minimize_energy'
        )
        elapsed = time.time() - start
        
        metrics = {
            'energy': getattr(solution, 'energy', -1),
            'min_distance': getattr(solution, 'min_distance', -1),
            'iterations': getattr(solution, 'iterations', -1),
            'time': round(elapsed, 3)
        }
        
        print(f"   ✅ Энергия: {metrics['energy']:.2f}")
        print(f"   📏 Min dist: {metrics['min_distance']:.2f}")
        print(f"   ⏱️  Время: {metrics['time']} с")
        return metrics, None
        
    except Exception as e:
        elapsed = time.time() - start
        print(f"   ❌ Ошибка: {str(e)[:100]}")
        return None, str(e)

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
safe_benchmark.py — БЕЗОПАСНЫЙ ТЕСТ МАСШТАБИРОВАНИЯ
Финальная версия: правильно извлекаем данные из solution.data

import os
import sys
import time
import csv
from pathlib import Path
from datetime import datetime

# ========== ПОРОГИ ПОД ТВОЁ ЖЕЛЕЗО ==========
RAM_MIN_FREE_GB = 0.8      # Минимум свободной RAM
SSD_MIN_FREE_GB = 5.0      # Минимум свободного места на SSD
PAUSE_BETWEEN_TESTS_SEC = 90  # Пауза между тестами (охлаждение)
# ============================================

try:
    import psutil
except ImportError:
    print("❌ Установи psutil: pip install psutil")
    sys.exit(1)

# Добавляем корень проекта в путь импорта
sys.path.insert(0, str(Path(__file__).parent.parent))

# ========== ИМПОРТЫ ==========
from architect import TopologicalArchitect
from architect.components import create_component_from_library
# =============================

def create_test_components(N):
    """Создаёт N компонентов через библиотеку architect"""
    components = []
    
    type_names = ['transmon_qubit', 'si_photonic_modulator', 'finfet_processor']
    charges = [1, -1, 0]
    
    for i in range(N):
        comp_type = type_names[i % len(type_names)]
        comp_id = f"{comp_type}_{i}"
        
        comp_dict = create_component_from_library(comp_type, comp_id)
        comp_dict['charge'] = charges[i % len(charges)]
        
        if 'qubit' in comp_type:
            comp_dict['frequency'] = 5.0 + i * 0.1
            comp_dict['anharmonicity'] = -0.3
        elif 'modulator' in comp_type:
            comp_dict['bandwidth'] = 50 + i * 2
        elif 'processor' in comp_type:
            comp_dict['cores'] = 2
        
        components.append(comp_dict)
    
    return components

def check_resources():
    """Проверка памяти и диска перед запуском"""
    ram = psutil.virtual_memory()
    ram_free = ram.available / (1024**3)
    ram_percent = ram.percent
    
    disk = psutil.disk_usage('C:\\')
    ssd_free = disk.free / (1024**3)
    ssd_percent = disk.percent
    
    issues = []
    if ram_free < RAM_MIN_FREE_GB:
        issues.append(f"RAM {ram_free:.2f} ГБ < {RAM_MIN_FREE_GB}")
    if ssd_free < SSD_MIN_FREE_GB:
        issues.append(f"SSD {ssd_free:.2f} ГБ < {SSD_MIN_FREE_GB}")
    
    return len(issues) == 0, issues, ram_free, ram_percent, ssd_free, ssd_percent

def run_test(N, grid_size=16):
    """Один прогон архитектора"""
    print(f"\n🚀 Тест N={N}")
    
    # Создаём компоненты
    components = create_test_components(N)
    
    # Создаём архитектор
    architect = TopologicalArchitect()
    
    start = time.time()
    try:
        # ФОРМИРУЕМ PROBLEM СЛОВАРЬ
        problem = {
            "name": f"benchmark_N{N}",
            "components": components,
            "grid_shape": [grid_size, grid_size, grid_size],
            "optimization_iters": 100,
            "target_metrics": {
                "min_distance": 5.0,
                "max_energy": 5000.0,
                "packing_coefficient": 0.2
            }
        }
        
        solution = architect.synthesize(problem)
        elapsed = time.time() - start
        
        # ИЗВЛЕКАЕМ ДАННЫЕ ИЗ solution.data
        result_data = solution.data if hasattr(solution, 'data') else {}
        metrics = {
            'energy': result_data.get('field_energy', result_data.get('energy', -1)),
            'min_distance': result_data.get('min_distance', -1),
            'iterations': result_data.get('iterations', result_data.get('n_iterations', -1)),
            'time': round(elapsed, 3)
        }

python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
safe_benchmark.py — БЕЗОПАСНЫЙ ТЕСТ МАСШТАБИРОВАНИЯ
Финальная версия: расстояние вычисляется из component_positions
"""

import os
import sys
import time
import csv
from pathlib import Path
from datetime import datetime

# ========== ПОРОГИ ПОД ТВОЁ ЖЕЛЕЗО ==========
RAM_MIN_FREE_GB = 0.8      # Минимум свободной RAM
SSD_MIN_FREE_GB = 5.0      # Минимум свободного места на SSD
PAUSE_BETWEEN_TESTS_SEC = 90  # Пауза между тестами (охлаждение)
# ============================================

try:
    import psutil
except ImportError:
    print("❌ Установи psutil: pip install psutil")
    sys.exit(1)

# Добавляем корень проекта в путь импорта
sys.path.insert(0, str(Path(__file__).parent.parent))

# ========== ИМПОРТЫ ==========
from architect import TopologicalArchitect
from architect.components import create_component_from_library
import numpy as np
# =============================

def create_test_components(N):
    """Создаёт N компонентов через библиотеку architect"""
    components = []
    
    type_names = ['transmon_qubit', 'si_photonic_modulator', 'finfet_processor']
    charges = [1, -1, 0]
    
    for i in range(N):
        comp_type = type_names[i % len(type_names)]
        comp_id = f"{comp_type}_{i}"
        
        comp_dict = create_component_from_library(comp_type, comp_id)
        comp_dict['charge'] = charges[i % len(charges)]
        
        if 'qubit' in comp_type:
            comp_dict['frequency'] = 5.0 + i * 0.1
            comp_dict['anharmonicity'] = -0.3
        elif 'modulator' in comp_type:
            comp_dict['bandwidth'] = 50 + i * 2
        elif 'processor' in comp_type:
            comp_dict['cores'] = 2
        
        components.append(comp_dict)
    
    return components

def check_resources():
    """Проверка памяти и диска перед запуском"""
    ram = psutil.virtual_memory()
    ram_free = ram.available / (1024**3)
    ram_percent = ram.percent
    
    disk = psutil.disk_usage('C:\\')
    ssd_free = disk.free / (1024**3)
    ssd_percent = disk.percent
    
    issues = []
    if ram_free < RAM_MIN_FREE_GB:
        issues.append(f"RAM {ram_free:.2f} ГБ < {RAM_MIN_FREE_GB}")
    if ssd_free < SSD_MIN_FREE_GB:
        issues.append(f"SSD {ssd_free:.2f} ГБ < {SSD_MIN_FREE_GB}")
    
    return len(issues) == 0, issues, ram_free, ram_percent, ssd_free, ssd_percent

def run_test(N, grid_size=16):
    """Один прогон архитектора"""
    print(f"\n🚀 Тест N={N}")
    
    # Создаём компоненты
    components = create_test_components(N)
    
    # Создаём архитектор
    architect = TopologicalArchitect()
    
    start = time.time()
    try:
        # ФОРМИРУЕМ PROBLEM СЛОВАРЬ
        problem = {
            "name": f"benchmark_N{N}",
            "components": components,
            "grid_shape": [grid_size, grid_size, grid_size],
            "optimization_iters": 100,
            "target_metrics": {
                "min_distance": 5.0,
                "max_energy": 5000.0,
                "packing_coefficient": 0.2
            }
        }
        
        solution = architect.synthesize(problem)
        elapsed = time.time() - start
        
        # ИЗВЛЕКАЕМ ДАННЫЕ ИЗ solution.data
        result_data = solution.data if hasattr(solution, 'data') else {}
        
        # Энергия
        energy = result_data.get('field_energy', -1)
        
        # ВЫЧИСЛЯЕМ МИНИМАЛЬНОЕ РАССТОЯНИЕ МЕЖДУ КОМПОНЕНТАМИ
        positions = result_data.get('component_positions', [])
        min_dist = -1.0
        if positions and len(positions) > 1:
            distances = []
            for i in range(len(positions)):
                for j in range(i+1, len(positions)):
                    d = np.linalg.norm(np.array(positions[i]) - np.array(positions[j]))
                    distances.append(d)
            min_dist = min(distances) if distances else -1.0
        
        metrics = {
            'energy': round(energy, 2),
            'min_distance': round(min_dist, 2),
            'iterations': result_data.get('iterations', result_data.get('n_iterations', -1)),
            'time': round(elapsed, 3)
        }
        
        print(f"   ✅ Энергия: {metrics['energy']:.2f}")
        print(f"   📏 Min dist: {metrics['min_distance']:.2f}")
        print(f"   ⏱️  Время: {metrics['time']} с")
        return metrics, None
        
    except Exception as e:
        elapsed = time.time() - start
        print(f"   ❌ Ошибка: {str(e)[:200]}")
        return None, str(e)

def main():
    print("\n" + "="*60)
    print("🔬 SPECTRAVORTEX — ЗАЩИЩЁННЫЙ ТЕСТ МАСШТАБИРОВАНИЯ")
    print("="*60)
    
    # Проверка ресурсов
    ok, issues, ram_free, ram_perc, ssd_free, ssd_perc = check_resources()
    print(f"\n🖥️  RAM: {ram_free:.2f} ГБ свободно ({ram_perc:.1f}%)")
    print(f"💾 SSD: {ssd_free:.2f} ГБ свободно ({ssd_perc:.1f}%)")
    
    if not ok:
        print(f"\n❌ Ресурсы недостаточны: {', '.join(issues)}")
        print("   Тесты отменены. Освободи место/память.")
        return
    
    print("\n✅ Ресурсы в норме. Начинаем тесты...")
    
    # ===== ТЕСТЫ =====
    test_N = [5, 10, 15, 20, 25]
    
    for N in test_N:
        if N != test_N[0]:
            print(f"\n⏸️  Пауза {PAUSE_BETWEEN_TESTS_SEC} с для охлаждения...")
            time.sleep(PAUSE_BETWEEN_TESTS_SEC)
        
        metrics, error = run_test(N)
    
    print("\n" + "="*60)
    print("✅ БЕНЧМАРК ЗАВЕРШЁН")
    print("="*60)

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n\n⏹️  Прервано пользователем")

3. С ЧЕМ СРАВНИТЬ (АНАЛОГИ)
3.1. Qiskit / IBM Quantum
Что делают: симуляция квантовых схем, оптимизация gate count.
Как размещают: вручную, либо фиксированная решётка.
Топология: линейная, решётка, HEA.
Минус: нет синтеза архитектуры. Есть placement, но он эвристический.
Наше преимущество: физический вариационный принцип, не перебор.

3.2. Synopsys / Cadence (EDA)
Что делают: автоматическое размещение миллионов транзисторов.
Как размещают: force-directed, simulated annealing, analytical placement.
Топология: 2D, фиксированные слои.
Минус: закрыто, $$$, нет квантовых компонентов.
Наше преимущество: 3D, гетерогенные компоненты (квант+фотоника+CMOS).

3.3. Ovito / LAMMPS (молекулярная динамика)
Что делают: симуляция атомов, кластеров, наночастиц.
Как размещают: потенциалы Леннард-Джонса, Кулона, EAM.
Топология: любая, но только классическая физика.
Минус: нет топологических зарядов, нет бигармоники.
Наше преимущество: единый математический аппарат для любых масштабов.

3.4. Gaussian / VASP (квантовая химия)
Что делают: расчёт электронной структуры, энергии связи.
Как размещают: DFT, HF, MP2.
Топология: координаты атомов.
Минус: O(N³), N≤100. Дорого.
Наше преимущество: O(N²) для сетки, масштабируется на тысячи компонентов.

1. ЧТО МЫ БЕРЁМ ЗА «КОНДЕНСАТ»?
В Noyron: нет конденсата. Там код, правила, геометрическое ядро PicoGK .
В топологической оптимизации: плотность материала, SIMP, метод конечных элементов .
В биомиметике Additive Rocket Corporation: фракталы, сосудистые сети, эволюция .

У нас: конденсат — это поле Ψ, в котором «плотность» = способность удерживать тепловой поток/давление/напряжение без разрушения.

Твоя подстановка:

SpectraVortex	Ракетный двигатель
Поле Ψ	Поле температур + напряжений + давлений
Вихрь (τ)	Очаг тепловыделения (камера сгорания)
Длина когерентности ξ	Толщина теплового пограничного слоя 
Заряд τ	Теплонапряжённость узла (Вт/м³ · МПа)
Размещение компонентов	Трассировка охлаждающих каналов
Конденсат = рабочее тело + материал стенки, связанные через Ψ.
В ракете нет разделения «конструкция» и «поток». Они решают одно уравнение тепломассопереноса . У нас — ∇⁴ψ = ρ. У них — div(λ·grad T) + Q = 0, связанное с гидравликой.

Принять за конденсат можно:

Теплопроводящий материал с фазовым переходом (регенеративное охлаждение)

Пористую среду для транспирации

Сам газ в камере, но тогда время жизни вихря — микросекунды. Нужна другая асимптотика.

Я предлагаю:
Конденсат = стенка камеры + охлаждающий тракт, рассматриваемые как единая бигармоническая среда.
Потому что именно в стенке замыкается баланс: пришло тепло от газового вихря — ушло в жидкость. Если не ушло — материал «коллапсирует» (плавится). Это и есть τ — способность отвести тепло без разрушения.

2. КАК ВЫГЛЯДИТ ДВИЖОК НА НАШЕЙ МАТЕМАТИКЕ?
Ядро — решатель ∇⁴T = q_v.
Не путать с обычной теплопроводностью (∇²T = 0). Бигармоника даёт изгибные моды изотерм. В стенке с каналами изотермы не гладкие — они огибают полости. Это требует четвёртого порядка.

Архитектура:

text
Вход: 
  - Геометрия (заготовка, envelope)
  - Тепловой поток от газа (карта q(x,y,z) из CFD)
  - Давление охладителя на входе
  - Материал (λ, σ_T, предел текучести)

Вихревая модель:
  - Каждый канал = вихрь с циркуляцией Γ ∝ ṁ (расход)
  - Положение каналов = координаты вихрей
  - Заряд τ = ṁ · c_p · ΔT_max (теплосъёмная способность)

Решатель:
  - Минимизация E = ∫|∇²T|² dV  (равномерное температурное поле)
  - При ограничениях: T < T_плавл, ΔP < P_насоса, расстояние между каналами > 0.5 мм
